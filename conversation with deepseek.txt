function importBulkEntries() {
    const eventId = document.getElementById('eventSelector').value;
    const text = document.getElementById('bulkText').value;
    
    if (!text.trim()) {
        alert('Please paste athlete data');
        return;
    }
    
    // Parse: name, class, sex, bib (tab or comma separated)
    const lines = text.split('\n').filter(l => l.trim());
    const athletes = lines.map(line => {
        const parts = line.split(/[\t,]+/).map(s => s.trim());
        return {
            name: parts[0],
            class: parts[1] || '',
            sex: parts[2] || 'M',
            bib: parts[3] || '',
            club: parts[4] || '',
            seasonBest: parts[5] || ''
        };
    });
    
    // First ensure all athletes exist in registry
    const registerPromises = athletes.map(a => 
        fetch('/api/athletes', {
            method: 'POST',
            headers: {'Content-Type': 'application/json'},
            body: JSON.stringify(a)
        }).then(r => r.json())
    );
    
    Promise.all(registerPromises)
        .then(results => {
            const athleteIds = results.map(r => r.id);
            // Now add them to the event
            return fetch('/api/entries/bulk', {
                method: 'POST',
                headers: {'Content-Type': 'application/json'},
                body: JSON.stringify({
                    eventId: eventId,
                    athleteIds: athleteIds
                })
            });
        })
        .then(r => r.json())
        .then(() => {
            alert(`✓ Successfully added ${athletes.length} athletes to the event!`);
            document.getElementById('bulkModal').style.display = 'none';
            document.getElementById('bulkText').value = '';
            loadEvent(eventId);
        })
        .catch(err => alert('Error: ' + err.message));
}

javascript
// Results export endpoint
app.get('/api/events/:id/results/export', (req, res) => {
    const eventId = req.params.id;
    const event = getEvent(eventId);
    const entries = getEntries(eventId);
    const marks = getMarks(eventId);
    const athletes = getAthletes();
    
    // Build results with rankings
    const results = entries.map(entry => {
        const athlete = athletes.find(a => a.id === entry.athleteId);
        const mark = marks.find(m => m.athleteId === entry.athleteId);
        return {
            name: athlete ? athlete.name : 'Unknown',
            class: athlete ? athlete.class : '',
            sex: athlete ? athlete.sex : '',
            bib: athlete ? athlete.bib : '',
            club: athlete ? athlete.club : '',
            mark: mark ? mark.mark : 'DNS',
            rank: mark && mark.rank ? mark.rank : '-',
            points: mark && mark.points ? mark.points : ''
        };
    });
    
    // Sort by rank
    results.sort((a, b) => {
        if (a.rank === '-') return 1;
        if (b.rank === '-') return -1;
        return a.rank - b.rank;
    });
    
    // Create CSV
    const header = 'Rank,Name,Class,Sex,Bib,Club,Mark,Points';
    const rows = results.map(r => 
        `${r.rank},${r.name},${r.class},${r.sex},${r.bib},${r.club},${r.mark},${r.points}`
    );
    const csv = [header, ...rows].join('\n');
    
    res.setHeader('Content-Type', 'text/csv');
    res.setHeader('Content-Disposition', `attachment; filename="${event.name}_results.csv"`);
    res.send(csv);
});

function searchAthletes() {
    const query = document.getElementById('lookupInput').value.toLowerCase().trim();
    if (query.length < 2) {
        document.getElementById('lookupResults').innerHTML = '<p style="color:#999;">Type at least 2 characters...</p>';
        return;

    fetch('/api/athletes')
        .then(r => r.json())
        .then(athletes => {
            const matches = athletes.filter(a => 
                a.name.toLowerCase().includes(query) || 
                a.bib === query ||
                a.class.toLowerCase().includes(query)
            ).slice(0, 20);
            
            if (matches.length === 0) {
                document.getElementById('lookupResults').innerHTML = '<p style="color:#999;">No athletes found</p>';
                return;
            }
            
            let html = '<table style="width:100%; border-collapse:collapse; margin-top:10px;">';

                html += `<tr style="border-bottom:1px solid #eee;">`;
                html += `<td style="padding:8px;">${a.name}</td>`;
                html += `<td style="padding:8px;">${a.class}</td>`;
                html += `<td style="padding:8px;">${a.bib || '-'}</td>`;
                html += `<td style="padding:8px;">${a.club || '-'}</td>`;

            html += '</table>';

const puppeteer = require('puppeteer');

// PDF Export - Start List
app.get('/api/events/:id/startlist/pdf', async (req, res) => {
    try {
        const eventId = req.params.id;
        const event = getEvent(eventId);
        const entries = getEntries(eventId);
        const athletes = getAthletes();
        const meet = getMeet();
        
        // Build start list data
        const startList = entries.map(entry => {
            const athlete = athletes.find(a => a.id === entry.athleteId);
            return {
                name: athlete ? athlete.name : 'Unknown',
                class: athlete ? athlete.class : '',
                bib: athlete ? athlete.bib : '',
                club: athlete ? athlete.club : '',
                seed: entry.seed || '',
                guide: athlete && ['T11','T12','F11','F12'].includes(athlete.class) ? 'Yes' : ''
            };
        });
        
        // Sort by seed time (fastest first for lane seeding)
        startList.sort((a, b) => {
            const aSeed = parseTime(a.seed) || Infinity;
            const bSeed = parseTime(b.seed) || Infinity;
            return aSeed - bSeed;
        });
        
        // Generate HTML
        const html = generateStartListPDF(event, startList, meet);
        
        // Launch puppeteer and generate PDF
        const browser = await puppeteer.launch({ headless: 'new' });
        const page = await browser.newPage();
        await page.setContent(html, { waitUntil: 'networkidle0' });
        
        const pdf = await page.pdf({
            format: 'A4',
            printBackground: true,
            margin: { top: '20mm', right: '20mm', bottom: '20mm', left: '20mm' }
        });
        
        await browser.close();
        
        res.setHeader('Content-Type', 'application/pdf');
        res.setHeader('Content-Disposition', `attachment; filename="${event.name}_StartList.pdf"`);
        res.send(pdf);
        
    } catch (error) {
        console.error('PDF generation error:', error);
        res.status(500).json({ error: error.message });
    }
});

// PDF Export - Results
app.get('/api/events/:id/results/pdf', async (req, res) => {
    try {
        const eventId = req.params.id;
        const event = getEvent(eventId);
        const entries = getEntries(eventId);
        const marks = getMarks(eventId);
        const athletes = getAthletes();
        const meet = getMeet();
        
        // Build results with rankings
        const results = entries.map(entry => {
            const athlete = athletes.find(a => a.id === entry.athleteId);
            const mark = marks.find(m => m.athleteId === entry.athleteId);
            return {
                name: athlete ? athlete.name : 'Unknown',
                class: athlete ? athlete.class : '',
                bib: athlete ? athlete.bib : '',
                club: athlete ? athlete.club : '',
                mark: mark ? mark.mark : 'DNS',
                rank: mark && mark.rank ? mark.rank : '-',
                points: mark && mark.points ? mark.points : '',
                guide: athlete && ['T11','T12','F11','F12'].includes(athlete.class) ? 'Yes' : ''
            };
        });
        
        // Sort by rank
        results.sort((a, b) => {
            if (a.rank === '-') return 1;
            if (b.rank === '-') return -1;
            return a.rank - b.rank;
        });
        
        const html = generateResultsPDF(event, results, meet);
        
        const browser = await puppeteer.launch({ headless: 'new' });
        const page = await browser.newPage();
        await page.setContent(html, { waitUntil: 'networkidle0' });
        
        const pdf = await page.pdf({
            format: 'A4',
            printBackground: true,
            margin: { top: '20mm', right: '20mm', bottom: '20mm', left: '20mm' }
        });
        
        await browser.close();
        
        res.setHeader('Content-Type', 'application/pdf');
        res.setHeader('Content-Disposition', `attachment; filename="${event.name}_Results.pdf"`);
        res.send(pdf);
        
    } catch (error) {
        console.error('PDF generation error:', error);
        res.status(500).json({ error: error.message });
    }
});

// Helper: Parse time string to seconds
function parseTime(timeStr) {
    if (!timeStr) return null;
    const str = String(timeStr).trim();
    
    // Try HH:MM:SS
    const hms = str.match(/^(\d+):(\d+):(\d+\.?\d*)$/);
    if (hms) return parseInt(hms[1]) * 3600 + parseInt(hms[2]) * 60 + parseFloat(hms[3]);
    
    // Try MM:SS.ss
    const ms = str.match(/^(\d+):(\d+\.?\d*)$/);
    if (ms) return parseInt(ms[1]) * 60 + parseFloat(ms[2]);
    
    // Try plain seconds
    const sec = parseFloat(str);
    if (!isNaN(sec)) return sec;
    
    return null;
}

// Export World Para Athletics format (WPA XML)
app.get('/api/events/:id/export/wpa', (req, res) => {
    const eventId = req.params.id;
    const event = getEvent(eventId);
    const entries = getEntries(eventId);
    const marks = getMarks(eventId);
    const athletes = getAthletes();
    const meet = getMeet();
    
    // Build WPA XML structure
    const results = entries.map(entry => {
        const athlete = athletes.find(a => a.id === entry.athleteId);
        const mark = marks.find(m => m.athleteId === entry.athleteId);
        return {
            ...athlete,
            mark: mark ? mark.mark : 'DNS',
            rank: mark && mark.rank ? mark.rank : '-'
        };
    });
    
    // Sort by rank
    results.sort((a, b) => {
        if (a.rank === '-') return 1;
        if (b.rank === '-') return -1;
        return a.rank - b.rank;
    });
    
    let xml = '<?xml version="1.0" encoding="UTF-8"?>\n';
    xml += '<WPAResults>\n';
    xml += `  <Competition>\n`;
    xml += `    <Name>${escapeXml(meet.name || 'Para Athletics')}</Name>\n`;
    xml += `    <Venue>${escapeXml(meet.venue || '')}</Venue>\n`;
    xml += `    <Date>${meet.date || ''}</Date>\n`;
    xml += `  </Competition>\n`;
    xml += `  <Event>\n`;
    xml += `    <Name>${escapeXml(event.name)}</Name>\n`;
    xml += `    <Discipline>${escapeXml(event.discipline)}</Discipline>\n`;
    xml += `    <Sex>${event.sex}</Sex>\n`;
    xml += `    <Round>${event.round || 'Final'}</Round>\n`;
    xml += `    <Status>${event.status || 'unofficial'}</Status>\n`;
    xml += `    <Results>\n`;
    
    results.forEach((athlete, index) => {
        xml += `      <Athlete rank="${athlete.rank}">\n`;
        xml += `        <Name>${escapeXml(athlete.name)}</Name>\n`;
        xml += `        <Bib>${athlete.bib || ''}</Bib>\n`;
        xml += `        <Class>${athlete.class || ''}</Class>\n`;
        xml += `        <Club>${escapeXml(athlete.club || '')}</Club>\n`;
        xml += `        <Mark>${athlete.mark || 'DNS'}</Mark>\n`;
        if (athlete.guide) {
            xml += `        <Guide>${escapeXml(athlete.guide)}</Guide>\n`;
        }
        xml += `      </Athlete>\n`;
    });
    
    xml += `    </Results>\n`;
    xml += `  </Event>\n`;
    xml += '</WPAResults>';
    
    res.setHeader('Content-Type', 'application/xml');
    res.setHeader('Content-Disposition', `attachment; filename="${event.name}_WPA.xml"`);
    res.send(xml);
});

function escapeXml(str) {
    if (!str) return '';
    return str.replace(/&/g, '&amp;')
              .replace(/</g, '&lt;')
              .replace(/>/g, '&gt;')
              .replace(/"/g, '&quot;')
              .replace(/'/g, '&apos;');
}

const QRCode = require('qrcode');

// Generate QR code for athlete lookup
app.get('/api/athletes/:id/qr', async (req, res) => {
    try {
        const athleteId = req.params.id;
        const athlete = getAthletes().find(a => a.id === athleteId);
        
        if (!athlete) {
            return res.status(404).json({ error: 'Athlete not found' });
        }
        
        // Generate QR code with athlete data URL
        const data = JSON.stringify({
            id: athlete.id,
            name: athlete.name,
            class: athlete.class,
            bib: athlete.bib,
            club: athlete.club
        });
        
        const qrImage = await QRCode.toDataURL(data, {
            errorCorrectionLevel: 'H',
            margin: 2,
            width: 300
        });
        
        // Return as image
        const base64Data = qrImage.replace(/^data:image\/png;base64,/, '');
        const imgBuffer = Buffer.from(base64Data, 'base64');
        
        res.setHeader('Content-Type', 'image/png');
        res.setHeader('Content-Disposition', `attachment; filename="${athlete.name}_QR.png"`);
        res.send(imgBuffer);
        
    } catch (error) {
        console.error('QR generation error:', error);
        res.status(500).json({ error: error.message });
    }
});

function generateId() {
    return Date.now().toString(36) + Math.random().toString(36).substr(2, 5);
}

// Parse time string to seconds
function parseTime(timeStr) {
    if (!timeStr) return null;
    const str = String(timeStr).trim();
    if (str === 'DNS' || str === 'DNF' || str === 'DQ' || str === 'NM') return null;
    
    const hms = str.match(/^(\d+):(\d+):(\d+\.?\d*)$/);
    if (hms) return parseInt(hms[1]) * 3600 + parseInt(hms[2]) * 60 + parseFloat(hms[3]);
    
    const ms = str.match(/^(\d+):(\d+\.?\d*)$/);
    if (ms) return parseInt(ms[1]) * 60 + parseFloat(ms[2]);
    
    const sec = parseFloat(str);
    if (!isNaN(sec)) return sec;
    
    return null;
}

// Format time from seconds
function formatTime(seconds) {
    if (seconds === null || seconds === undefined || !isFinite(seconds)) return '';
    if (seconds >= 3600) {
        const h = Math.floor(seconds / 3600);
        const m = Math.floor((seconds % 3600) / 60);
        const s = (seconds % 60).toFixed(2);
        return `${h}:${String(m).padStart(2, '0')}:${String(s).padStart(5, '0')}`;
    } else if (seconds >= 60) {
        const m = Math.floor(seconds / 60);
        const s = (seconds % 60).toFixed(2);
        return `${m}:${String(s).padStart(5, '0')}`;
    } else {
        return seconds.toFixed(2);
    }
}

// Escape XML
function escapeXml(str) {
    if (!str) return '';
    return str.replace(/&/g, '&amp;')
              .replace(/</g, '&lt;')
              .replace(/>/g, '&gt;')
              .replace(/"/g, '&quot;')
              .replace(/'/g, '&apos;');
}

app.get('/api/events/:id/startlist', (req, res) => {
    const eventId = req.params.id;
    const event = getEvents().find(e => e.id === eventId);
    if (!event) return res.status(404).json({ error: 'Event not found' });
    
    const entries = getEntries().filter(e => e.eventId === eventId);
    const athletes = getAthletes();
    const reference = getReference();
    const guideClasses = reference.guideClasses || [];
    
    const startList = entries.map(entry => {
        const athlete = athletes.find(a => a.id === entry.athleteId);
        return {
            ...entry,
            athlete: athlete || null,
            hasGuide: athlete ? guideClasses.includes(athlete.class) : false
        };
    }).filter(s => s.athlete);
    
    // Sort by seed (fastest first for track events)
    const isTrack = event.discipline.includes('m') || event.discipline.includes('km');
    if (isTrack) {
        startList.sort((a, b) => {
            const aSeed = parseTime(a.athlete?.seed) || Infinity;
            const bSeed = parseTime(b.athlete?.seed) || Infinity;
            return aSeed - bSeed;
        });
    }
    
    res.json({ event, startList });
});

app.get('/api/events/:id/results', (req, res) => {
    const eventId = req.params.id;
    const event = getEvents().find(e => e.id === eventId);
    if (!event) return res.status(404).json({ error: 'Event not found' });
    
    const entries = getEntries().filter(e => e.eventId === eventId);
    const marks = getMarks().filter(m => m.eventId === eventId);
    const athletes = getAthletes();
    const reference = getReference();
    const guideClasses = reference.guideClasses || [];
    
    // Build results
    const results = entries.map(entry => {
        const athlete = athletes.find(a => a.id === entry.athleteId);
        const mark = marks.find(m => m.athleteId === entry.athleteId);
        return {
            entryId: entry.id,
            athleteId: entry.athleteId,
            name: athlete ? athlete.name : 'Unknown',
            class: athlete ? athlete.class : '',
            sex: athlete ? athlete.sex : '',
            bib: athlete ? athlete.bib : '',
            club: athlete ? athlete.club : '',
            mark: mark ? mark.mark : 'DNS',
            hasGuide: athlete ? guideClasses.includes(athlete.class) : false,
            guide: athlete && guideClasses.includes(athlete.class) ? 'Yes' : ''
        };
    });
    
    // Sort and rank
    const isTrack = event.discipline.includes('m') || event.discipline.includes('km');
    const isField = ['Shot Put', 'Discus', 'Javelin', 'Club Throw', 'Long Jump', 'Triple Jump', 'High Jump', 'Pole Vault'].includes(event.discipline);
    const isCombined = event.discipline === 'Pentathlon';
    
    results.forEach(r => {
        if (r.mark === 'DNS' || r.mark === 'DNF' || r.mark === 'DQ' || r.mark === 'NM') {
            r.value = null;
            r.isDNS = true;
        } else {
            r.value = parseTime(r.mark);
            r.isDNS = false;
        }
    });
    
    // Sort: valid marks first, then DNS/DNF/DQ/NM
    results.sort((a, b) => {
        if (a.isDNS && b.isDNS) return 0;
        if (a.isDNS) return 1;
        if (b.isDNS) return -1;
        if (isTrack) return a.value - b.value; // Fastest first
        if (isField || isCombined) return b.value - a.value; // Furthest first
        return 0;
    });
    
    // Assign ranks
    let rank = 1;
    results.forEach((r, index) => {
        if (r.isDNS) {
            r.rank = '-';
        } else {
            // Check for tie
            if (index > 0 && !results[index-1].isDNS && results[index-1].value === r.value) {
                r.rank = results[index-1].rank;
            } else {
                r.rank = rank;
            }
            rank++;
        }
    });
    
    res.json({ event, results });
});

// ---------- WPA EXPORT ----------
app.get('/api/events/:id/export/wpa', (req, res) => {
    const eventId = req.params.id;
    const event = getEvents().find(e => e.id === eventId);
    if (!event) return res.status(404).json({ error: 'Event not found' });
    
    const entries = getEntries().filter(e => e.eventId === eventId);
    const marks = getMarks().filter(m => m.eventId === eventId);
    const athletes = getAthletes();
    const meet = getMeet();
    
    const results = entries.map(entry => {
        const athlete = athletes.find(a => a.id === entry.athleteId);
        const mark = marks.find(m => m.athleteId === entry.athleteId);
        return {
            ...athlete,
            mark: mark ? mark.mark : 'DNS',
            rank: mark && mark.rank ? mark.rank : '-'
        };
    }).filter(a => a);
    
    results.sort((a, b) => {
        if (a.rank === '-') return 1;
        if (b.rank === '-') return -1;
        return a.rank - b.rank;
    });
    
    let xml = '<?xml version="1.0" encoding="UTF-8"?>\n';
    xml += '<WPAResults>\n';
    xml += `  <Competition>\n`;
    xml += `    <Name>${escapeXml(meet.name || 'Para Athletics')}</Name>\n`;
    xml += `    <Venue>${escapeXml(meet.venue || '')}</Venue>\n`;
    xml += `    <Date>${meet.date || ''}</Date>\n`;
    xml += `  </Competition>\n`;
    xml += `  <Event>\n`;
    xml += `    <Name>${escapeXml(event.name)}</Name>\n`;
    xml += `    <Discipline>${escapeXml(event.discipline)}</Discipline>\n`;
    xml += `    <Sex>${event.sex}</Sex>\n`;
    xml += `    <Round>${event.round || 'Final'}</Round>\n`;
    xml += `    <Status>${event.status || 'unofficial'}</Status>\n`;
    xml += `    <Results>\n`;
    
    results.forEach(athlete => {
        xml += `      <Athlete rank="${athlete.rank}">\n`;
        xml += `        <Name>${escapeXml(athlete.name)}</Name>\n`;
        xml += `        <Bib>${athlete.bib || ''}</Bib>\n`;
        xml += `        <Class>${athlete.class || ''}</Class>\n`;
        xml += `        <Club>${escapeXml(athlete.club || '')}</Club>\n`;
        xml += `        <Mark>${athlete.mark || 'DNS'}</Mark>\n`;
        if (athlete.guide) {
            xml += `        <Guide>${escapeXml(athlete.guide)}</Guide>\n`;
        }
        xml += `      </Athlete>\n`;
    });
    
    xml += `    </Results>\n`;
    xml += `  </Event>\n`;
    xml += '</WPAResults>';
    
    res.setHeader('Content-Type', 'application/xml');
    res.setHeader('Content-Disposition', `attachment; filename="${event.name}_WPA.xml"`);
    res.send(xml);
});

                container.innerHTML = html;
                
            } catch (error) {
                container.innerHTML = renderEmptyState('⚠️', 'Error', error.message);
            }

        async function renderAthletesView() {
            const container = document.getElementById('mainContent');
            container.innerHTML = renderLoading();
            
            try {
                const athletes = await fetchAPI('/api/athletes');
                const meet = await fetchAPI('/api/meet');
                
                if (athletes.length === 0) {
                    container.innerHTML = renderEmptyState(
                        '👤',
                        'No Athletes',
                        'Import athletes from the admin panel.'
                    );
                    return;
                }
                
                // Group by class
                const grouped = {};
                athletes.forEach(a => {
                    const cls = a.class || 'Unknown';
                    if (!grouped[cls]) grouped[cls] = [];
                    grouped[cls].push(a);
                });
                
                const sortedClasses = Object.keys(grouped).sort();
                
                let html = `
                    <div class="refresh-control">
                        <span>${athletes.length} athletes</span>
                        <button class="refresh-btn" onclick="renderAthletesView()">↻ Refresh</button>
                    </div>
                `;
                
                // Search box
                html += `
                    <div class="card" style="padding:12px;">
                        <input type="text" id="athleteSearch" placeholder="🔍 Search athletes..." 
                               style="width:100%; padding:10px; background:var(--bg); border:1px solid var(--border); border-radius:8px; color:var(--text); font-size:14px;"
                               oninput="filterAthletes()">
                    </div>
                `;
                
                html += `<div id="athleteList">`;
                
                sortedClasses.forEach(cls => {
                    html += `
                        <div class="card" style="padding:12px 16px;">
                            <div class="card-title">Class ${cls}</div>
                            ${grouped[cls].map(a => renderAthleteCard(a)).join('')}
                        </div>
                    `;
                });
                
                html += `</div>`;
                container.innerHTML = html;
                
            } catch (error) {
                container.innerHTML = renderEmptyState('⚠️', 'Error', error.message);
            }
        }
        
        // Athlete search filter
        function filterAthletes() {
            const query = document.getElementById('athleteSearch').value.toLowerCase();
            const items = document.querySelectorAll('#athleteList .card');
            items.forEach(card => {
                const text = card.textContent.toLowerCase();
                card.style.display = text.includes(query) ? 'block' : 'none';
            });
        }
        
        // ---------- Navigation ----------
        function switchView(view) {
            currentView = view;
            
            // Update nav
            document.querySelectorAll('.nav-item').forEach(el => {
                el.classList.toggle('active', el.dataset.view === view);
            });
            
            // Clear auto-refresh for non-live views
            if (view !== 'live' && autoRefreshInterval) {
                clearInterval(autoRefreshInterval);
                autoRefreshInterval = null;
            }
            
            // Render view
            switch(view) {
                case 'events':
                    renderEventsView();
                    break;
                case 'live':
                    renderLiveView();
                    if (!autoRefreshInterval) {
                        autoRefreshInterval = setInterval(renderLiveView, 5000);
                    }
                    break;
                case 'results':
                    renderResultsView();
                    break;
                case 'athletes':
                    renderAthletesView();
                    break;
            }
        }
        
        function selectEvent(eventId) {
            selectedEventId = eventId;
            if (currentView === 'live') {
                renderLiveView();
            } else {
                switchView('live');
            }
        }
        
        function refreshAll() {
            switchView(currentView);
        }
        
        // ---------- Init ----------
        document.addEventListener('DOMContentLoaded', () => {
            renderEventsView();
            
            // Set up PWA installation
            let deferredPrompt;
            window.addEventListener('beforeinstallprompt', (e) => {
                e.preventDefault();
                deferredPrompt = e;
                // Show install button if needed
            });
        });
        
        // Handle visibility change - refresh when coming back
        document.addEventListener('visibilitychange', () => {
            if (!document.hidden) {
                refreshAll();
            }
        });
    </script>

# Para Athletics Competition Management System

Complete system for managing Para Athletics competitions with mobile app interface.

## Features

### Admin Features
- ✅ Dynamic event scheduling with days/sessions
- ✅ Athlete registry with CSV import
- ✅ Event entries with class/sex filtering
- ✅ Results entry with automatic ranking
- ✅ Guide class detection (T11/T12/F11/F12)
- ✅ Lane seeding (fastest to lane 4)
- ✅ Summary editor for each event

### Export Features
- 📄 PDF Start Lists
- 📄 PDF Results with rankings
- 📊 CSV Results Export
- 🌐 World Para Athletics XML Export
- 📱 QR codes for each athlete

### Display Features
- 📺 Live results display
- 🏟️ Scoreboard mode for projectors
- 📱 Mobile app interface (PWA)

The mobile interface is a Progressive Web App (PWA) that can be installed on phones:
1. Open http://localhost:3000/mobile.html on your phone
2. For iOS: tap Share → Add to Home Screen
3. For Android: tap ⋮ → Add to Home Screen

### Mobile Features
- 📋 Event list with status
- 📺 Live results with auto-refresh
- 🏆 Medal standings
- 👤 Athlete lookup

## Data Files

All data is stored in `server/data/`:
- `athletes.json` - Master athlete registry
- `entries.json` - Which athletes in which events
- `marks.json` - Performance marks
- `events.json` - Event definitions
- `meet.json` - Competition details
- `reference.json` - Rules (classes, restrictions)

## Workflow

1. **Setup**: Configure meet details in Admin
2. **Import Athletes**: Paste from Excel in Athletes page
3. **Create Events**: Add events in Admin
4. **Enter Athletes**: Add athletes to events in Entries page
5. **Record Marks**: Enter results in Entries page
6. **Publish**: Results auto-rank, links go live
7. **Export**: PDFs, CSV, WPA XML

## Requirements

- Node.js 16+
- npm
